function updatePlantOfTheDay() {
 const now = new Date();
 const start = new Date(now.getFullYear(), 0, 0);
 const dayOfYear = Math.floor((now - start) / 86400000);
 const dateKey = `${now.getFullYear()}-${String(now.getMonth()+1).padStart(2,'0')}-${String(now.getDate()).padStart(2,'0')}`;
 // Stable date-based selection; the plant changes each day and cycles through available guides.
 let hash = 0;
 for (const char of dateKey) hash = (hash * 31 + char.charCodeAt(0)) >>> 0;
 const chosenKey = plantDayKeys.length ? plantDayKeys[hash % plantDayKeys.length] : 'lemon';
 const entry = data[chosenKey] || data.lemon;
 $('plantDayDate').textContent = `A fresh little growing idea for ${now.toLocaleDateString(undefined,{month:'long',day:'numeric'})}.`;
 $('plantDayBadge').textContent = `${entry.icon} ${entry.name}`;
 $('plantDayIcon').textContent = entry.icon;
 $('plantDayTitle').textContent = entry.name;
 $('plantDaySummary').textContent = entry.summary;
 $('plantDayGuideBtn').onclick = () => openGuide(entry);
}
